技术实现

渲染管线 · 性能优化 · 代码架构

// DriveModeController.cs — 驾驶模式切换控制器
// 挂在场景中的 "HMIController" GameObject 上，供 JS 通过 SendMessage 调用
using UnityEngine;

public class DriveModeController : MonoBehaviour
{
    public Material cockpitEmissionMat;   // 座舱氛围灯材质
    public Light[] ambientLights;         // 氛围灯灯组

    // 供网页 JS 调用：unityInstance.SendMessage("HMIController", "SetDriveMode", "Sport")
    public void SetDriveMode(string mode)
    {
        switch (mode)
        {
            case "Comfort":
                ApplyMode(Color.cyan, 0.6f);
                break;
            case "Sport":
                ApplyMode(new Color(1f, 0.42f, 0f), 1.2f);
                break;
            case "Eco":
                ApplyMode(Color.green, 0.35f);
                break;
        }
    }

    void ApplyMode(Color c, float intensity)
    {
        cockpitEmissionMat.SetColor("_EmissionColor", c * intensity);
        foreach (var l in ambientLights) { l.color = c; l.intensity = intensity; }
    }
}
